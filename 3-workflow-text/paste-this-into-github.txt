name: Build installer
on:
  push:
    branches: [main, master]
  workflow_dispatch:
permissions:
  contents: write
jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Install
        run: npm install
      - name: Build the exe files
        run: npm run dist
        env:
          CSC_IDENTITY_AUTO_DISCOVERY: "false"
      - name: Publish a download
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: scxfyy launcher build ${{ github.run_number }}
          files: |
            dist/scxfyy-launcher.exe
            dist/scxfyy-launcher-setup.exe
